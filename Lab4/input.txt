const long BIT_MASK = 0xFFFF;
const int HEX_CONSTANT = 0x1A;

short bitwise_operations(short a, short b) {
    short and_result = a & b;
    short or_result = a | b;
    short xor_result = a ^ b;
    
    return and_result | or_result | xor_result;
}

long complex_calculation(long value) {
    long temp = value;
    int counter = 0;
    
    while ((temp & BIT_MASK) != 0) {
        temp = temp >> 4;
        counter = counter + 1;
    }
    
    return counter;
}

const long long_value = 0xFFFFFFFFL;

int main() {
    short num1 = 0x1010;
    short num2 = 0x1100;
    
    short result = bitwise_operations(num1, num2);
    
    const long long_value = 0xFFFFFFFFL;
    long calc_result = complex_calculation(long_value);
    
    return (result + calc_result);
}
