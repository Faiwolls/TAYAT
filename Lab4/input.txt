int main() {
	long x = 222L; //Десятичный лонг с суффиксом
	long hex_x = 0xFF222L; //Шестнадцатеричный с суффиксом
	long zero = 0L; //Длинный ноль

	long small_x = 222l; //Десятичный лонг с маленьким суффиксом
	long small_hex_x = 0xFF222l; //Шестнадцатеричный с маленьким суффиксом
	long small_zero = 0l; //Длинный ноль с маленьким суффиксом
}
